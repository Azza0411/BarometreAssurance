SELECT COUNT(DISTINCT c.id) AS nb_compagnies
FROM documents d
JOIN sources s ON s.id = d.source_id AND s.nom = 'CMF'
JOIN societes c ON c.id = d.societe_id
JOIN kpi_values kv ON kv.document_id = d.id
WHERE d.annee = 2024
  AND kv.kpi = 'Ratio combiné (%)'
  AND kv.valeur_nombre BETWEEN 0.1 AND 500
  AND kv.valeur_nombre < 100;
